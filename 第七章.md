## 1.判断下列表达式的真假：

- a. `100 > 3 && 'a' > 'c'`：假（0）。`'a' > 'c'` 不成立。
- b. `100 > 3 || 'a' > 'c'`：真（1）。左侧已经成立。
- c. `!(100 > 3)`：假（0）。`100 > 3` 为真，取反后为假。

## 2.根据条件构造表达式：

- a. `number` 等于或大于 90，但小于 100：`number >= 90 && number < 100`
- b. `ch` 不是字符 `q` 或 `k`：`ch != 'q' && ch != 'k'`
- c. `number` 在 1～9 之间（包括 1 和 9），但不是 5：`number >= 1 && number <= 9 && number != 5`
- d. `number` 不在 1～9 之间：`number < 1 || number > 9`

## 3.简化并改正程序中的关系表达式：

```c
#include <stdio.h>
int main(void)
{
    int weight, height;
    scanf("%d , weight, height);
    if (weight < 100 && height > 64)
        if (height >= 72)
            printf("You are very tall for your weight.\n");
        else if (height < 72 && > 64)
            printf("You are tall for your weight.\n");
    else if (weight > 300 && !(weight <= 300)
             && height < 48)
        if (!(height >= 48))
            printf(" You are quite short for your weight.\n");
    else
        printf("Your weight is ideal.\n");
    return 0;
}
```

改正并简化后：

```c
#include <stdio.h>

int main(void)
{
    int weight, height;

    scanf("%d %d", &weight, &height);
    if (weight < 100 && height > 64)
    {
        if (height >= 72)
            printf("You are very tall for your weight.\n");
        else
            printf("You are tall for your weight.\n");
    }
    else if (weight > 300 && height < 48)
        printf(" You are quite short for your weight.\n");
    else
        printf("Your weight is ideal.\n");

    return 0;
}
```

修改要点：

- `scanf("%d , weight, height);` 缺少地址符和引号，应为 `scanf("%d %d", &weight, &height);`。
- `height < 72 && > 64` 缺少左操作数；外层已经保证 `height > 64`，进入 `else` 时只需判断 `height < 72`。
- `weight > 300 && !(weight <= 300)` 与 `weight > 300` 等价，可以删去多余部分。
- `if (!(height >= 48))` 与外层的 `height < 48` 重复，可以直接输出。
- 增加花括号，避免 `else` 与内层 `if` 错误配对。

## 5.下面的程序将打印什么：

```c
#include <stdio.h>

int main(void)
{
    int num;

    for (num = 1; num <= 11; num++)
    {
        if (num % 3 == 0)
            putchar('$');
        else
            putchar('*');
        putchar('#');
        putchar('%');
    }
    putchar('\n');

    return 0;
}
```

```text
*#%*#%$#%*#%*#%$#%*#%*#%$#%*#%*#%
```

`num` 能被 3 整除时输出 `$#%`，否则输出 `*#%`，共 11 组。

## 6.下面的程序将打印什么：

```c
#include <stdio.h>

int main(void)
{
    int i = 0;

    while (i < 3)
    {
        switch (i++)
        {
            case 0:
                printf("fat ");
            case 1:
                printf("hat ");
            case 2:
                printf("cat ");
            default:
                printf("Oh no!");
        }
        putchar('\n');
    }

    return 0;
}
```

```text
fat hat cat Oh no!
hat cat Oh no!
cat Oh no!
```

每个 `case` 都没有 `break`，因此会继续执行后面的分支。


## 8.下面的程序将打印什么：

```c
#include <stdio.h>

int main(void)
{
    int age = 20;

    while (age++ <= 65)
    {
        if ((age % 20) == 0)
            printf("You are %d.  Here is a raise.\n", age);
        if (age = 65)
            printf("You are %d.  Here is your gold watch.\n", age);
    }

    return 0;
}
```

程序会无限次打印：

```text
You are 65.  Here is your gold watch.
```

原因是 `if (age = 65)` 使用的是赋值而不是相等比较 `==`。每次执行都把 `age` 设为 65，条件恒为真；下一轮 `while (age++ <= 65)` 又把 `age` 增至 66，随后再次被赋值为 65，循环无法结束。

### 2.打印字符及其 ASCII 码，每行 8 组

```c
#include <stdio.h>

int main(void)
{
    int ch;
    int count = 0;

    printf("请输入文本：");
    while ((ch = getchar()) != '#')
    {
        if (ch == '\n')
            printf("\\n:%d ", ch);
        else if (ch == '\t')
            printf("\\t:%d ", ch);
        else
            printf("%c:%d ", ch, ch);

        count++;
        if (count % 8 == 0)
            printf("\n");
    }
    printf("\n");

    return 0;
}
```


### 4.用 `if else` 替换字符并计数

```c
#include <stdio.h>

int main(void)
{
    int ch;
    int replacements = 0;

    printf("请输入文本：");
    while ((ch = getchar()) != '#')
    {
        if (ch == '.')
        {
            putchar('!');
            replacements++;
        }
        else if (ch == '!')
        {
            printf("!!");
            replacements++;
        }
        else
            putchar(ch);
    }

    return 0;
}
```

### 6.统计 `ei` 出现的次数

```c
#include <stdio.h>

int main(void)
{
    int ch, prev = 0;
    int count = 0;

    printf("请输入文本：");
    while ((ch = getchar()) != '#')
    {
        if (prev == 'e' && ch == 'i')
            count++;
        prev = ch;
    }
    printf("ei 出现了 %d 次。\n", count);

    return 0;
}
```




### 9.输出小于或等于输入值的所有素数

```c
#include <stdio.h>

int main(void)
{
    int n, i, j, is_prime;

    do {
        printf("请输入一个正整数：");
        if (scanf("%d", &n) != 1)
        {
            while (getchar() != '\n')
                ;
            n = 0;
        }
    } while (n <= 0);

    for (i = 2; i <= n; i++)
    {
        is_prime = 1;
        for (j = 2; j * j <= i; j++)
        {
            if (i % j == 0)
            {
                is_prime = 0;
                break;
            }
        }
        if (is_prime)
            printf("%d ", i);
    }
    printf("\n");

    return 0;
}
```
